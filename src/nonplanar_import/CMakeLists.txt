add_executable(nonplanar_import_worker main.cpp)
add_executable(nonplanar_stl_audit audit.cpp)
add_dependencies(nonplanar_stl_audit nonplanar_import_worker)
target_compile_definitions(nonplanar_stl_audit PRIVATE NPTOP_WORKER_NAME="nonplanar_import_worker${CMAKE_EXECUTABLE_SUFFIX}")
foreach(_target nonplanar_import_worker nonplanar_stl_audit)
  set_target_properties(${_target} PROPERTIES MACOSX_BUNDLE FALSE)
  target_link_libraries(${_target} libslic3r ${CMAKE_DL_LIBS})
  if(APPLE)
    target_link_libraries(${_target} "-liconv -framework IOKit" "-framework CoreFoundation" -lc++)
  endif()
  if(WIN32)
    set(_worker_configs ${CMAKE_CONFIGURATION_TYPES})
    if(NOT _worker_configs)
        set(_worker_configs "${CMAKE_BUILD_TYPE}")
    endif()
    foreach(_cfg IN LISTS _worker_configs)
        if(_cfg STREQUAL "Debug")
            orcaslicer_copy_dlls(${_target} "${_cfg}" "d" _worker_dlls)
        else()
            orcaslicer_copy_dlls(${_target} "${_cfg}" "" _worker_dlls)
        endif()
    endforeach()
  endif()
endforeach()
