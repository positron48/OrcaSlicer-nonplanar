add_executable(nonplanar_import_worker main.cpp)
set_target_properties(nonplanar_import_worker PROPERTIES MACOSX_BUNDLE FALSE)
target_link_libraries(nonplanar_import_worker libslic3r)
if(APPLE)
    target_link_libraries(nonplanar_import_worker "-liconv -framework IOKit" "-framework CoreFoundation" -lc++)
endif()
if(WIN32)
    set(_worker_configs ${CMAKE_CONFIGURATION_TYPES})
    if(NOT _worker_configs)
        set(_worker_configs "${CMAKE_BUILD_TYPE}")
    endif()
    foreach(_cfg IN LISTS _worker_configs)
        if(_cfg STREQUAL "Debug")
            orcaslicer_copy_dlls(nonplanar_import_worker "${_cfg}" "d" _worker_dlls)
        else()
            orcaslicer_copy_dlls(nonplanar_import_worker "${_cfg}" "" _worker_dlls)
        endif()
    endforeach()
endif()
