add_executable(nonplanar_analysis_worker main.cpp)
set_target_properties(nonplanar_analysis_worker PROPERTIES MACOSX_BUNDLE FALSE)
target_link_libraries(nonplanar_analysis_worker libslic3r ${CMAKE_DL_LIBS})
if(APPLE)
  target_link_libraries(nonplanar_analysis_worker "-liconv -framework IOKit" "-framework CoreFoundation" -lc++)
endif()
if(WIN32)
  set(_worker_configs ${CMAKE_CONFIGURATION_TYPES})
  if(NOT _worker_configs)
    set(_worker_configs "${CMAKE_BUILD_TYPE}")
  endif()
  foreach(_cfg IN LISTS _worker_configs)
    if(_cfg STREQUAL "Debug")
      orcaslicer_copy_dlls(nonplanar_analysis_worker "${_cfg}" "d" _worker_dlls)
    else()
      orcaslicer_copy_dlls(nonplanar_analysis_worker "${_cfg}" "" _worker_dlls)
    endif()
  endforeach()
endif()
